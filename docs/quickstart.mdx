## 1. Install peer dependencies

```bash
npm install @ricsam/react-mui-calendar @mui/material @emotion/react @emotion/styled date-fns
```

## 2. Add the provider

Use `CalendarThemeProvider` for a standalone calendar. If your application already has a MUI theme, see [Theming](/theming).

```tsx
import { CalendarThemeProvider } from "@ricsam/react-mui-calendar";

export function Root() {
  return (
    <CalendarThemeProvider>
      <App />
    </CalendarThemeProvider>
  );
}
```

## 3. Render a controlled calendar

```tsx
import { endOfDay, startOfDay } from "date-fns";
import { useState } from "react";
import { MonthCalendar, type CalendarEvent } from "@ricsam/react-mui-calendar";

const initialEvents: CalendarEvent[] = [
  {
    title: "Launch",
    start: startOfDay(new Date()),
    end: endOfDay(new Date()),
    styling: { bg: "#4338ca", textColor: "#fff" },
    canEdit: true,
  },
];

export function Calendar() {
  const [events, setEvents] = useState(initialEvents);

  return (
    <div style={{ height: 640 }}>
      <MonthCalendar
        events={events}
        startOfMonth={new Date()}
        onMoveEvent={(event, start, end) => {
          setEvents((current) =>
            current.map((item) =>
              item === event ? { ...item, start, end } : item,
            ),
          );
        }}
      />
    </div>
  );
}
```

<Tip>
  Give `MonthCalendar` a parent with an explicit height. It measures available
  height to determine how many event rows fit before showing the overflow menu.
</Tip>
